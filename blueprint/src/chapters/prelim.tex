\begin{definition}[Independence Number]
    \label{def:indepNum}
    \lean{SimpleGraph.indepNum}
    \leanok
    $I(G)$, the independence number of the graph $G$, 
    is the maximum number of points of $G$ that can be chosen so that no two are joined by an edge.
\end{definition}

\begin{definition}[Clique Number]
    \label{def:cliqueNum}
    \lean{SimpleGraph.cliqueNum}
    \leanok
    $C(G)$, the clique number of the graph $G$, is the maximum number of points in any complete subgraph of $G$.
\end{definition}
