\chapter{Ramsey Theory}

\begin{definition}[$(x,y)$-graphs]
    \label{def:xyGraph}
    \lean{SimpleGraph.isXYGraph}
    \leanok
    % \uses{def:indepNum, def:cliqueNum}
    $G$ is an $(x,y)$-graph if $x > C(G)$ and $y > I(G)$.
\end{definition}

\begin{definition}[Ramsey Number]
    \label{def:ramseynumber}
    \lean{SimpleGraph.RamseyOld}
    \leanok
    \uses{def:xyGraph}
    $$
    R(x,y) = \max \{ n \in \mathbb{N} \mid \exists G \text{, } G \text{ is an} (x, y)\text{-graph } \land \text{ } |V(G)| = n \}
    $$
\end{definition}

\begin{lemma}[Graver Lemma 1]
    \label{l:Lemma1}
    \lean{SimpleGraph.Lemma₁}
    \leanok
    \uses{def:xyGraph}
    $G$ is an $(x,y)$-graph if and only if its complement $G^c$ is a $(y,x)$-graph.
\end{lemma}
\begin{proof}
    \leanok
\end{proof}

\begin{lemma}[Graver Lemma 2]
    \label{l:Lemma2}
    \lean{SimpleGraph.Lemma₂}
